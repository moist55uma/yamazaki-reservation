/* ======================================== POST受信 ======================================== */ function doPost(e) { try { const p = e.parameter || {}; const date = String(p.date || '').trim(); const time = String(p.time || '').trim(); const people = String(p.people || '').trim(); const name = String(p.name || '').trim(); const phone = String(p.phone || '').trim(); const email = String(p.email || '').trim(); const memo = String(p.memo || '').trim(); const source = normalizeSource( p.source || '' ); /* ======================================== 必須項目チェック ======================================== */ if ( !date || !time || !people || !name || !phone || !email ) { return response( 'error', '必須項目が入力されていません' ); } /* ======================================== 営業日チェック ======================================== */ if ( !OPEN_DATES.includes(date) ) { return response( 'error', 'この日は予約を受け付けていません' ); } /* ======================================== 日本時間 ======================================== */ const now = new Date(); if ( !validTimes.includes(time) ) { return response( 'error', 'この時間は予約できません' ); } /* ======================================== 当日予約締切 ======================================== */ if ( date === todayString ) { const currentTime = Utilities.formatDate( now, 'Asia/Tokyo', 'HH:mm' ); /* ランチ 10:30締切 */ if ( LUNCH_TIMES.includes(time) && currentTime >= '10:30' ) { return response( 'error', '本日のランチ予約受付は終了しました' ); } /* ディナー 16:00締切 */ if ( DINNER_TIMES.includes(time) && currentTime >= '16:00' ) { return response( 'error', '本日のディナー予約受付は終了しました' ); } } /* ======================================== スプレッドシート ======================================== */ const ss = SpreadsheetApp .getActiveSpreadsheet(); let sheet = ss.getSheetByName( SHEET_NAME ); if (!sheet) { sheet = ss.getSheets()[0]; } /* ======================================== 見出し ======================================== */ if ( sheet.getLastRow() === 0 ) { sheet.appendRow([ '受付日時', '来店日', '時間', '人数', '名前', '電話番号', 'メール', '備考', '流入元', '予約状況' ]); sheet.setFrozenRows(1); } /* ======================================== 予約を台帳に登録 ======================================== */ sheet.appendRow([ new Date(), date, time, people, name, /* 電話番号はあとから文字列で設定 */ '', email, memo, source, '未確認' ]); const lastRow = sheet.getLastRow(); /* ======================================== 電話番号を文字列として保存 090の先頭0を消さない ======================================== */ const phoneCell = sheet.getRange( lastRow, 6 ); phoneCell .setNumberFormat('@'); phoneCell .setValue(phone); /* ======================================== 店舗通知メール ======================================== */ const shopEmail = Session .getEffectiveUser() .getEmail(); const shopSubject = `【山咲WEB予約】${date} ${time} ${name}様`; const shopBody = `海鮮 山咲に新しい予約リクエストが入りました。 ━━━━━━━━━━━━━━━━ 来店日：${date} 時間：${time} 人数：${people} お名前：${name} 電話番号：${phone} メール：${email} 備考： ${memo || 'なし'} 流入元：${source} ━━━━━━━━━━━━━━━━ ※この時点ではまだ予約確定ではありません。 内容をご確認のうえ、 お客様へ予約確定のご連絡をお願いします。`; /* ======================================== お客様への自動返信 ======================================== */ const customerSubject = '【海鮮 山咲】予約リクエストを受け付けました / Reservation Request Received'; const customerBody = `${name} 様 この度は海鮮 山咲へ ご予約リクエストをいただき、 ありがとうございます。 下記の内容で受け付けました。 ━━━━━━━━━━━━━━━━ 来店日：${date} 時間：${time} 人数：${people} ━━━━━━━━━━━━━━━━ 【重要】 こちらは「予約リクエスト受付」のメールです。 この時点では、 まだ予約は確定しておりません。 店舗からの確認連絡をもって 予約確定となります。 予約時間を10分過ぎてもご来店がない場合は、 自動キャンセルとさせていただく場合がございます。 ご都合が悪くなった場合は、 事前にキャンセルのご連絡をお願いいたします。 ────────────── English ────────────── Thank you for your reservation request at Kaisendon Yamasaki. If your plans change, please contact us in advance to cancel your reservation. ────────────── 繁體中文 ────────────── 感謝您向海鮮 山咲提出訂位申請。 我們已收到以下訂位資訊： 日期：${date} 時間：${time} 人數：${people} 【重要】 此郵件僅表示我們已收到您的訂位申請。 目前尚未正式確認訂位。 收到餐廳的確認通知後， 您的訂位才正式成立。 若超過預約時間10分鐘仍未到店， 且未事先聯絡， 我們可能會取消您的訂位。 若您的行程有變， 請提前與我們聯絡取消訂位。 海鮮 山咲 Kaisendon Yamasaki 越後湯沢 / Echigo-Yuzawa`; /* ======================================== メール送信 メールだけ失敗しても 台帳の予約データは消さない ======================================== */ try { MailApp.sendEmail( shopEmail, shopSubject, shopBody ); } catch (mailError) { console.error( '店舗メール送信エラー：' + mailError ); } try { MailApp.sendEmail( email, customerSubject, customerBody ); } catch (mailError) { console.error( 'お客様メール送信エラー：' + mailError ); } /* ======================================== 正常終了 ======================================== */ return response( 'success', '予約リクエストを受け付けました' ); } catch (error) { console.error( error ); return response( 'error', String(error) ); } } /* ======================================== 流入元 ======================================== */ function normalizeSource(source) { const s = String(source) .trim() .toLowerCase(); if ( s === 'ig' || s === 'instagram' ) { return 'Instagram'; } if ( s === 'threads' || s === 'thread' ) { return 'Threads'; } if ( s === 'google' || s === 'gmb' ) { return 'Google'; } if (!s) { return '不明'; } return source; } /* ======================================== レスポンス ======================================== */ function response( status, message ) { return ContentService .createTextOutput( JSON.stringify({ status:status, message:message }) ) .setMimeType( ContentService.MimeType.JSON ); }
